Import-Module ActiveDirectory

$GlobalGroupPath = "OU=Globaux,OU=Groupes,DC=griffon,DC=local"
$DomainLocalPath = "OU=Domaines_Locaux,OU=Groupes,DC=griffon,DC=local"

$GlobalGroups = @(
    "GG_DIRECTION",
    "GG_RH",
    "GG_FINANCE",
    "GG_BE",
    "GG_PRODUCTION",
    "GG_IT"
)

$DomainLocalGroups = @(
    "DL_DIRECTION_RW",
    "DL_RH_RW",
    "DL_FINANCE_RW",
    "DL_BE_RW",
    "DL_PRODUCTION_RW",
    "DL_IT_ADMIN"
)

foreach ($Group in $GlobalGroups) {
    if (-not (Get-ADGroup -Filter "Name -eq '$Group'" -ErrorAction SilentlyContinue)) {
        New-ADGroup -Name $Group -GroupScope Global -GroupCategory Security -Path $GlobalGroupPath
    }
}

foreach ($Group in $DomainLocalGroups) {
    if (-not (Get-ADGroup -Filter "Name -eq '$Group'" -ErrorAction SilentlyContinue)) {
        New-ADGroup -Name $Group -GroupScope DomainLocal -GroupCategory Security -Path $DomainLocalPath
    }
}

Add-ADGroupMember -Identity "DL_DIRECTION_RW" -Members "GG_DIRECTION"
Add-ADGroupMember -Identity "DL_RH_RW" -Members "GG_RH"
Add-ADGroupMember -Identity "DL_FINANCE_RW" -Members "GG_FINANCE"
Add-ADGroupMember -Identity "DL_BE_RW" -Members "GG_BE"
Add-ADGroupMember -Identity "DL_PRODUCTION_RW" -Members "GG_PRODUCTION"
Add-ADGroupMember -Identity "DL_IT_ADMIN" -Members "GG_IT"

Write-Host "Groupes AGDLP créés et chaînage configuré."
